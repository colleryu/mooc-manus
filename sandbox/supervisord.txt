[unix_http_server]
file=/tmp/supervisor.sock ;socket文件路径
chmod=0770                ;socket文件权限，770表示文件所有者和同组用户有读/写/执行权限，其他
chown=root:root           ;socket文件的所有者和组


[supervisord]
logfile=/dev/stdout       ;主日志路径
logfile_maxbytes=0        ;不限制日志大小(用完即弃无需 考虑)
loglevel=info             ;日志等级
pidfile=/tmp/supervisord.pid ;pid文件路径
nodaemon=true             ;设置为True表示在前台运行(docker前台)，方便docker捕捉日志
minfds=1024               ;打开文件描述的最小值
minprocs=200              ;最小进程
autoshutdown=true         ;所有服务停止后自动退出

[rpcinterface:supervisor]
;启用supervisor的远程过程调用接口(RPC),可以通过接口控制supervisor
supervisor.rpcinterface_factory = supervisor.rpcinterface:make_main_rpcinterface

[supervisorctl]
;告诉suoervisorctl工具取哪里链接supervisord服务
serverurl=unix:///tmp/supervisor.sock
